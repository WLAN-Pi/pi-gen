# For more options and information see

# Uncomment some or all of these to enable the optional hardware interfaces
# i2c_arm required for WLAN Pi Pro RTC + battery fuel gauge (and M4+ EEPROM)
dtparam=i2c_arm=on
#dtparam=i2s=on
dtparam=spi=on

# Enable audio (loads snd_bcm2835)
dtparam=audio=on

# Additional overlays and parameters are documented
# /boot/firmware/overlays/README

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Automatically load overlays for detected DSI displays
display_auto_detect=1

# Enable DRM VC4 V3D driver. Also loads snd-soc-hdmi-codec, which
# wlanpi-model relies on to identify the M4 (lsmod | grep hdmi)
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Use firmware framebuffer for display across all platforms
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

# Enable heartbeat
dtparam=act_led_trigger=heartbeat

[pi4]
kernel=wlanpi-kernel8.img

[cm4]
otg_mode=1
kernel=wlanpi-kernel8.img
dtoverlay=dwc2,dr_mode=host

# Disable CM4's Wi-Fi and BT radio if they are present
dtoverlay=disable-wifi
dtoverlay=disable-bt

# WLAN Pi Pro hardware (toggled per-platform by wlanpi-config-at-startup.sh)
dtoverlay=gpio-fan,gpiopin=26,temp=70000
dtoverlay=i2c-rtc,pcf85063a,addr=0x51
dtoverlay=battery_gauge

[pi5]
kernel=wlanpi-kernel_2712.img

[cm5]
dtoverlay=dwc2,dr_mode=host
kernel=wlanpi-kernel_2712.img

[all]
